    s = np.sin(2*np.pi*f*t)*np.exp(-t*5) + 0.4*np.sin(2*np.pi*3.93*f*t)*np.exp(-t*14) + 0.15*np.sin(2*np.pi*7.9*f*t)*np.exp(-t*20)
    e = np.minimum(1, t/0.002)
    return s * e * 0.8

def voice_piano(f, d):
    t = Tdur(d)
    s = np.zeros_like(t)
    for k in range(1, 8):
        det = 1 + 0.0004*k*k
        s += np.sin(2*np.pi*f*k*det*t) * np.exp(-t*(1.6+k*0.9)) / k
    e = np.minimum(1, t/0.003)
    # soft pedal sustain: longer release
    e *= np.exp(-t*0.8) + 0.3*np.exp(-t*3)
    return s * e * 0.5

def voice_brass(f, d):
    t = Tdur(d)
    vib = 1 + 0.006*np.sin(2*np.pi*5.5*t)
    s = np.zeros_like(t)
    for k in range(1, 10):
        s += np.sin(2*np.pi*f*k*vib*t) / (k**1.25)
    e = np.minimum(1, t/0.03) * np.exp(-t*2.2)
    return s * e * 0.35

def voice_organ(f, d):
    t = Tdur(d)
    s = np.sin(2*np.pi*f*t)*1.0 + np.sin(2*np.pi*2*f*t)*0.5 + np.sin(2*np.pi*3*f*t)*0.3 + np.sin(2*np.pi*4*f*t)*0.2
    e = np.minimum(1, t/0.01) * np.minimum(1, (d - t)/0.05).clip(0,1)
    return s * e * 0.3

def voice_pad_saw(f, d):
    t = Tdur(d)
    # detuned saws bandlimited additive
    max_h = int(SR/(2*max(f,1)))
    max_h = min(max_h, 16)
    s = np.zeros_like(t)
    for det in (1.0, 1.004, 0.996):
        for k in range(1, max_h+1):
            s += np.sin(2*np.pi*f*k*det*t) / k
    s *= 0.08
    e = np.minimum(1, t/0.3) * np.minimum(1, (d - t)/0.3).clip(0,1)
    # lowpass via simple one-pole
    # apply lfilter lowpass 2kHz
    b = [0.1]; a=[1, -0.9]
    s = lfilter([0.2], [1, -0.8], s)
    return s * e

def voice_strings(f, d):
    t = Tdur(d)
    max_h = int(SR/(2*max(f,1)))
    max_h = min(max_h, 12)
    s = np.zeros_like(t)
    for det in (1.0, 1.007, 0.993):
        for k in range(1, max_h+1):
            s += np.sin(2*np.pi*f*k*det*t) / (k**1.1) * (0.7 if k==1 else 1)
    e = np.minimum(1, t/0.15) * np.minimum(1, (d - t)/0.2).clip(0,1)
    s = lfilter([0.15], [1, -0.85], s)
    return s * e * 0.18

def voice_glock(f, d):
    t = Tdur(d)
    s = np.sin(2*np.pi*f*t) + 0.5*np.sin(2*np.pi*2.76*f*t)*np.exp(-t*4) + 0.3*np.sin(2*np.pi*5.4*f*t)*np.exp(-t*8)
    e = np.minimum(1, t/0.002) * np.exp(-t*2.5)
    return s * e * 0.4

def karplus_strong(f, d, brightness=0.5):
    n = int(round(d*SR))
    p = max(2, int(round(SR/f)))
    # excite with noise burst
    buf = rng.uniform(-1, 1, p) * 0.5
    # simple lowpass in loop via lfilter later vectorized using lfilter? Use iterative but with numba? Use lfilter approach:
    # We can generate via lfilter of impulse train: KS is comb filter. Use efficient method: create excitation then filter
    # Simple iterative but p small (~100) and n ~ 48000*2 = 96000, loop in python would be slow.
    # Use vectorized lfilter: y[n]= x[n] + 0.996*0.5*(y[n-p]+y[n-p-1])
    # This is IIR comb; we can implement with lfilter using sparse coefficients? Use iterative numba-like python loop but with numpy for each p block?
    # We'll do block loop in python but with numpy per block: still O(n) python loop ~ 1000 iterations per note worst, okay for limited notes.
    # Number of pluck notes ~ maybe 100, each 1 sec => 48000 samples => 480 iterations per note => 48000 python iterations total ~ maybe 50000 loops -> too slow.
    # Alternative: use scipy lfilter with custom a,b
    # KS transfer: y = x + g*(y_delay)
    # We can implement using lfilter with a = [1, 0,..., -g*0.5, -g*0.5]
    # Create a array length p+1
    g = 0.996
    a = np.zeros(p+1)
    a[0]=1
    a[p-1] = -g*0.5
    a[p] = -g*0.5
    # excitation: noise burst first p samples then zeros
    x = np.zeros(n)
    x[:p] = buf
    # apply lfilter
    y = lfilter([1], a, x)
    # envelope
    t = np.arange(n)/SR
    y *= np.exp(-t*1.5) * 0.8
    return y

def drum_kick(d=0.3):
    t = Tdur(d)
    f = 60 + 80*np.exp(-t*30)
    s = np.sin(2*np.pi*np.cumsum(f)/SR) * np.exp(-t*12)
    # click
    s[:int(0.005*SR)] += np.hanning(int(0.005*SR))*0.5
    return s * 0.9

def drum_snare(d=0.25, brushed=False):
    t = Tdur(d)
    if brushed:
        n = rng.uniform(-1,1,len(t)) * np.exp(-t*30) * 0.4
        # bandpass via simple
        n = lfilter([0.2,0,-0.2],[1,-0.5], n)
        tone = np.sin(2*np.pi*180*t)*np.exp(-t*20)*0.3
        return (n+tone)*0.6
    else:
        n = rng.uniform(-1,1,len(t)) * np.exp(-t*40) * 0.5
        tone = np.sin(2*np.pi*180*t)*np.exp(-t*25)*0.4
        return (n+tone)*0.6

def drum_hat(d=0.15, open_h=False):
    t = Tdur(d)
    n = rng.uniform(-1,1,len(t))
    # highpass
    n = lfilter([1,-1],[1,-0.9], n)
    e = np.exp(-t*(15 if open_h else 60))
    return n * e * 0.25

def drum_clap(d=0.3):
    t = Tdur(d)
    n = rng.uniform(-1,1,len(t))
    e = np.exp(-t*30) * (1+0.5*np.sin(2*np.pi*30*t))
    s = n*e*0.4
    # add filtered
    s = lfilter([0.3],[1,-0.7], s)
    return s

def sfx_synth(kind, t_abs=None):
    # tuned to D major where pitched
    if kind == 'tick':
        t = Tdur(0.04)
        return np.sin(2*np.pi*3200*t)*np.exp(-t*160)*0.35
    if kind == 'click':
        t = Tdur(0.03)
        return rng.uniform(-1,1,len(t))*np.exp(-t*250)*0.35
    if kind == 'chime':
        t = Tdur(1.2)
        f0 = midi_to_hz(74) # D5
        s = np.sin(2*np.pi*f0*t)*np.exp(-t*1.5) + 0.5*np.sin(2*np.pi*2.76*f0*t)*np.exp(-t*3) + 0.3*np.sin(2*np.pi*5.4*f0*t)*np.exp(-t*5)
        return s*0.28
    if kind == 'thump':
        t = Tdur(0.3)
        f = 50 + 70*np.exp(-t*30)
        return np.sin(2*np.pi*np.cumsum(f)/SR)*np.exp(-t*12)*0.7
    if kind == 'beep':
        t = Tdur(0.15)
        f = midi_to_hz(74) # D5, will be transposed for countdown via caller
        return np.sin(2*np.pi*f*t)*np.minimum(1,(0.15-t)/0.02).clip(0,1)*0.28
    if kind == 'pop':
        t = Tdur(0.12)
        f = 400 + 900*np.exp(-t*40)
        return np.sin(2*np.pi*np.cumsum(f)/SR)*np.exp(-t*30)*0.45
    if kind == 'whoosh':
        t = Tdur(0.7)
        # soft airy tone not noise blast
        f = 300 + 100*np.sin(2*np.pi*0.5*t)
        s = np.sin(2*np.pi*f*t) * (np.sin(np.pi*t/0.7)**2) * 0.25
        # add filtered noise low
        n = rng.uniform(-1,1,len(t))*0.05 * np.sin(np.pi*t/0.7)**2
        n = lfilter([0.1],[1,-0.9], n)
        return s+n
    return np.zeros(int(0.02*SR))

def chord_notes(root_midi, deg, voicing='triad'):
    # return midi notes for chord
    # major scale triad: degrees 0,2,4 etc but need chord quality
    # Use diatonic triads in major: I major, ii minor, iii minor, IV major, V major, vi minor, vii dim
    # For simplicity use major/minor based on degree
    # Map deg 0:I,1:ii,2:iii,3:IV,4:V,5:vi,6:vii
    # We'll build triad by stacking thirds in scale
    r = deg_to_midi(root_midi, deg)
    # third
    third_deg = deg+2
    fifth_deg = deg+4
    t = deg_to_midi(root_midi, third_deg)
    f = deg_to_midi(root_midi, fifth_deg)
    # adjust octave to be in sensible register: root around C3, third/fifth above
    # Ensure chord voiced close
    return [r, t, f]

def bus_compress(x, thresh=0.25, ratio=2.0, attack=0.01, release=0.1):
    env = np.zeros_like(x)
    a_a = np.exp(-1/(attack*SR))
    a_r = np.exp(-1/(release*SR))
    e = 0
    for i in range(len(x)):
        v = abs(x[i])
        if v > e:
            e = a_a*e + (1-a_a)*v
        else:
            e = a_r*e + (1-a_r)*v
        env[i]=e
    gain = np.ones_like(x)
    over = env > thresh
    gain[over] = (thresh + (env[over]-thresh)/ratio) / (env[over]+1e-9)
    gain = np.clip(gain, 0.5, 1.0)
    gain = lfilter([0.01],[1,-0.99], gain)
    return x * gain
