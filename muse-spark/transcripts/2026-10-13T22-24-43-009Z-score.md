#!/usr/bin/env python3
"""Score a portal film from its plan: music synthesized in numpy, no samples, no network.

  node render.cjs <film> --plan plan.json
  python3 score.py plan.json score.wav

Each scene gets its own voice and tempo (scene.music = {voice, bpm, root}); its music starts when that scene takes the
frame (push: when the portal fills; pull: when the pull begins), on a downbeat. One four-bar phrase (the film's motif)
passes from voice to voice so the film feels like one piece. Pushes get a soft rising swell; scene.sfx entries
({u, kind}) land at act start + u. Sections are levelled to the same loudness, and the master is limited to -1 dBFS.

voices: piano, music_box, vibes, marimba, pluck, chip, pad, brass, organ
sfx:    tick, click, chime, whoosh, thump, beep, pop
"""
import json, sys, wave
import numpy as np

SR = 48000
plan = json.load(open(sys.argv[1]))
out = sys.argv[2] if len(sys.argv) > 2 else 'score.wav'
rng = np.random.default_rng(11)
TOTAL = plan['total'] + 1.0
N = int(TOTAL * SR)
L = np.zeros(N); R = np.zeros(N)

def T(d): return np.arange(int(d * SR)) / SR
def hz(m): return 440.0 * 2 ** ((m - 69) / 12)
def env(t, a=0.005, decay=3.0): return np.minimum(1, t / max(a, 1e-4)) * np.exp(-t * decay)
def put(buf_l, buf_r, sig, t0, gain=1.0, pan=0.0):
    i = int(round(t0 * SR))
    if i >= len(buf_l) or i + len(sig) <= 0: return
    if i < 0: sig, i = sig[-i:], 0
    sig = sig[:len(buf_l) - i]; a = (pan + 1) * np.pi / 4
    buf_l[i:i + len(sig)] += sig * gain * np.cos(a) * 1.414; buf_r[i:i + len(sig)] += sig * gain * np.sin(a) * 1.414

# ---------- voices: f (Hz), d (s) -> mono signal ----------
def piano(f, d):   t = T(d); return sum(np.sin(2*np.pi*f*k*(1+.0004*k*k)*t)*np.exp(-t*(1.6+k*.9))/k for k in range(1, 8)) * np.minimum(1, t/.003) * .7
def music_box(f, d): t = T(d); return (np.sin(2*np.pi*f*t) + .35*np.sin(2*np.pi*2.01*f*t)*np.exp(-t*3) + .18*np.sin(2*np.pi*4.17*f*t)*np.exp(-t*8)) * env(t, .002, 3.2)
def vibes(f, d):   t = T(d); return (np.sin(2*np.pi*f*t) + .25*np.sin(2*np.pi*3.99*f*t)*np.exp(-t*6)) * env(t, .003, 1.6) * (1 + .3*np.sin(2*np.pi*5.8*t))
def marimba(f, d): t = T(d); return (np.sin(2*np.pi*f*t)*np.exp(-t*5) + .4*np.sin(2*np.pi*3.93*f*t)*np.exp(-t*14)) * np.minimum(1, t/.002)
def pluck(f, d):
    n = int(d*SR); p = max(2, int(SR/f)); buf = rng.uniform(-1, 1, p); o = np.empty(n)
    for i in range(n): j = i % p; o[i] = buf[j]; buf[j] = .996 * .5 * (buf[j] + buf[(j+1) % p])
    return o * .8
def chip(f, d):    t = T(d); return np.where((f*t) % 1 < .25, 1., -1.) * env(t, .002, 5) * .28
def pad(f, d):     t = T(d); s = sum(np.sin(2*np.pi*f*k*det*t)/k for det in (1, 1.004, .996) for k in range(1, 7)); return s * np.minimum(1, t/.3) * np.minimum(1, (d-t)/.3).clip(0) * .12
def brass(f, d):   t = T(d); v = 1 + .006*np.sin(2*np.pi*5.5*t); return sum(np.sin(2*np.pi*f*k*v*t)/k**1.25 for k in range(1, 10)) * np.minimum(1, t/.03) * np.exp(-t*2.2) * .35
def organ(f, d):   t = T(d); return sum(np.sin(2*np.pi*f*k*t)*a for k, a in ((1, 1), (2, .5), (3, .3), (4, .2))) * np.minimum(1, t/.01) * np.minimum(1, (d-t)/.05).clip(0) * .3
VOICES = dict(piano=piano, music_box=music_box, vibes=vibes, marimba=marimba, pluck=pluck, chip=chip, pad=pad, brass=brass, organ=organ)

def sfx(kind):
    if kind == 'tick':  t = T(.04); return np.sin(2*np.pi*3200*t)*np.exp(-t*160)*.5
    if kind == 'click': t = T(.03); return rng.uniform(-1, 1, len(t))*np.exp(-t*250)*.5
    if kind == 'chime': t = T(1.6); return sum(np.sin(2*np.pi*1318*k*t)*np.exp(-t*(2+k))/k for k in (1, 2.76, 5.4)) * .35
    if kind == 'whoosh':
        t = T(.7); x = rng.uniform(-1, 1, len(t)); y = np.convolve(x, np.ones(40)/40, 'same'); return y * np.sin(np.pi*t/.7)**2 * .9
    if kind == 'thump': t = T(.3); f = 50 + 70*np.exp(-t*30); return np.sin(2*np.pi*np.cumsum(f)/SR)*np.exp(-t*12)
    if kind == 'beep':  t = T(.15); return np.sin(2*np.pi*1046*t)*np.minimum(1, (.15-t)/.02)*.3
    if kind == 'pop':   t = T(.12); f = 400 + 900*np.exp(-t*40); return np.sin(2*np.pi*np.cumsum(f)/SR)*np.exp(-t*30)*.5
    return np.zeros(1)

# ---------- the motif: four bars, (scale degree, beat, length in beats) over I-vi-IV-V ----------
MAJOR = [0, 2, 4, 5, 7, 9, 11]
def deg(root, d): o, k = divmod(d, 7); return root + 12*o + MAJOR[k]
MOTIF = [(4, 0, 1), (5, 1, 1), (7, 2, 2), (4, 4, .5), (5, 4.5, .5), (4, 5, 1), (2, 6, 2), (0, 8, 1), (2, 9, 1), (4, 10, 2), (5, 12, 1), (4, 13, 1), (1, 14, 2)]
CHORDS = [0, 5, 3, 4]

S = plan['shots']; bounds = [0.0]
for a, b in zip(S, S[1:]): bounds.append(a['end'] if a['out'] == 'push' else b['start'])
bounds.append(plan['total'] + 0.6)
sections = []
for i, sh in enumerate(S):
    m = sh.get('music') or {}; voice = VOICES.get(m.get('voice', 'piano'), piano); bpm = m.get('bpm', 92); root = m.get('root', 60)
    t0, t1 = bounds[i], bounds[i + 1]; beat = 60 / bpm
    sl, sr = np.zeros(N), np.zeros(N)
    nb = int((t1 - t0) / beat) + 1
    for b in range(nb):                                   # bass + chord pulse on the beat grid that starts at t0
        bar = (b // 4) % 4; ch = CHORDS[bar]; t = t0 + b * beat
        if b % 4 == 0: put(sl, sr, pad(hz(deg(root - 12, ch)), beat * 4), t, .7)
        if b % 2 == 0: put(sl, sr, piano(hz(deg(root - 24, ch)), beat * 1.8) * .6, t, .55, -.2)
    for rep in range(nb // 16 + 1):                       # the motif, repeated per 16 beats
        for d, at, ln in MOTIF:
            t = t0 + (rep * 16 + at) * beat
            if t < t1 - 0.05: put(sl, sr, voice(hz(deg(root, d)), max(.25, ln * beat * 1.4)), t, .8, .15)
    i0, i1, k = int(t0 * SR), min(N, int(t1 * SR)), int(.06 * SR)   # 60 ms fades at the section edges, hard musical entries
    fade = np.zeros(N); fade[i0:i1] = 1
    a = fade[i0:i0 + k]; a *= np.linspace(0, 1, k)[:len(a)]
    b = fade[max(i0, i1 - k):i1]; b *= np.linspace(1, 0, k)[-len(b):] if len(b) else 1
    sl *= fade; sr *= fade
    rms = np.sqrt(np.mean(sl[i0:i1] ** 2 + sr[i0:i1] ** 2) / 2) + 1e-9
    g = 0.12 / rms * (1.25 if i == len(S) - 1 else 1)       # level the sections; the last one sits a little prouder
    L += sl * g; R += sr * g
    for e in sh.get('sfx', []): put(L, R, sfx(e['kind']), e['t'], .6)
    if sh['out'] == 'push':                               # swell into the object
        d = sh['handover']; t = T(d); x = np.convolve(rng.uniform(-1, 1, len(t)), np.ones(60)/60, 'same')
        put(L, R, x * (t/d)**2 * .5, sh['end'] - d, 1)

if N > SR: L[-SR:] *= np.linspace(1, 0, SR); R[-SR:] *= np.linspace(1, 0, SR)
peak = max(np.abs(L).max(), np.abs(R).max()) + 1e-9
g = min(1.0, 10 ** (-1 / 20) / peak)                      # limit to -1 dBFS
st = (np.stack([L, R], 1) * g * 32767).astype('<i2')
with wave.open(out, 'wb') as w: w.setnchannels(2); w.setsampwidth(2); w.setframerate(SR); w.writeframes(st.tobytes())
print(f'{out}: {TOTAL:.2f} s, {len(S)} sections, peak {20*np.log10(peak*g):.1f} dBFS')

def bandlimited_pulse(f, d, duty=0.25):
    t = Tdur(d)
    # additive bandlimited pulse: sum harmonics up to Nyquist
    # pulse with duty d has harmonics amplitude sin(pi*k*duty)/(pi*k) etc but approximate with additive
    # limit harmonics to SR/(2*f)
    max_h = int(SR/(2*max(f,1))) 
    max_h = min(max_h, 24)
    s = np.zeros_like(t)
    for k in range(1, max_h+1):
        # Fourier of pulse 0-1
        coeff = (2/(np.pi*k)) * np.sin(np.pi*k*duty) if k!=0 else duty
        # alternative: use square-like
        s += coeff * np.sin(2*np.pi*f*k*t)

    e = np.minimum(1, t/0.005) * np.exp(-t*2.0)

# simple RMS compressor
def bus_compress(x, thresh=0.25, ratio=2.0, attack=0.01, release=0.1):
    # envelope follower
    env = np.zeros_like(x)
    a_a = np.exp(-1/(attack*SR))
    a_r = np.exp(-1/(release*SR))
    e = 0
    for i in range(len(x)):
        v = abs(x[i])
        if v > e:
            e = a_a*e + (1-a_a)*v
        else:
            e = a_r*e + (1-a_r)*v
        env[i]=e
    gain = np.ones_like(x)
    over = env > thresh
    gain[over] = (thresh + (env[over]-thresh)/ratio) / (env[over]+1e-9)
    gain = np.clip(gain, 0.5, 1.0)
    # smooth gain
    gain = lfilter([0.01],[1,-0.99], gain)
    return x * gain
